	- 19, 21페이지처럼 그림을 그려서 설명하라.

```c
#include <stdio.h>
int main(void)
{
	int x;
	char name[10]; 
	char blood_type; 
	printf("이름을 입력하시오:");
	scanf("%s", name); 
	printf("혈액형을 입력하시오:");
	scanf("%c", &blood_type);
	printf("이름: %s\n", name);
	printf("혈액형: %c\n", blood_type);
	return 0;
}
```

{
	int x;
	char name[10];
	char blood_type;
	printf("이름을 입력하시오: ");
	scanf("%s", name);
	getchar();
	printf("혈액형을 입력하시오: ");
	scanf("%c", &blood_type);
	printf("이름: %s\n", name);
	printf("혈액형: %c\n", blood_type);
	return 0;
}
