- 다음 코드의 문제점을 설명하고 포인터를 이용하여 문제를 해결하시오.
  - 배열의 이름은 해당 배열의 첫 번째 원소의 주소를 뜻하지만 상수이기 때문에 변수인 포인터에 할당하여 사용해야 함
```c
#include <stdio.h>
int main(void)
{
	int a[] = { 10, 20, 30 }, i, sum = 0;
	int* pA = a;

	for (i = 0; i < 3; i++)
		sum += *pA++;
	printf("sum:%d\n", sum);
	return 0;
}
```
